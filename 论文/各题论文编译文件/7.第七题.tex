	\subsection{题目重述}
		\begin{compactenum}
			\item 设计一个RC无源低通滤波器，分别构建仿真电路和实测电路，滤除题6[1]d)中的高频信号。
			\item 设计一个RC无源高通滤波器，分别构建仿真电路和实测电路，滤除题6[1]d)中的低频信号。
			\item 搭建实测电路，分别为上述低通实测电路引入一个20mV正弦信号，调整信号频率，在0.1Hz-1MHz 范围内测试30个以上关键频率点的输出波形，记录输出信号幅值变化，用MATLAB描点绘制Bode图，用术语描述滤波器的频率特性。
		\end{compactenum}
	\subsection{题目分析与实施方案}
		\subsubsection{第一问}
			第一问要求设计RC无源低通滤波器，最常见的RC滤波器是由一个电阻和一个电容串联分压构成的一阶滤波器，因此考虑在本题使用这个电路。\par
			电路确定后，通过计算无源网络的传递函数，得到对应相频特性函数。将需要滤除的频率代入特性曲线的截止频率，便可以推算出对应电路参数，并根据实际情况，选用相关精度的标准阻值电阻。\par
			然后，使用Multisim软件仿真对应电路，将题6[1]d)中的输出信号输入搭建的RC无源低通滤波器上，并连接示波器。观察示波器波形，判断高频信号是否被滤除。
		\subsubsection{第二问}
			第二问要求设计RC无源高通滤波器，最常见的RC滤波器是由一个电容和一个电阻串联分压构成的一阶滤波器，因此考虑在本题使用这个电路。\par
			电路确定后，通过计算无源网络的传递函数，得到对应相频特性函数。将需要滤除的频率代入特性曲线的截止频率，便可以推算出对应电路参数，并根据实际情况，选用相关精度的标准阻值电阻。\par
			然后，使用Multisim软件仿真对应电路，将题6[1]d)中的输出信号输入搭建的RC无源高通滤波器上，并连接示波器。观察示波器波形，判断低频信号是否被滤除。
		\subsubsection{第三问}
			根据第一问搭建的电路，向RC无源低通滤波器等对数频率，分别测试输出信号的幅值变化三十组。\par
			使用MATLAB的plot函数，描点绘制出测试得到的参数，以得到相频伯德图曲线。\par
			根据绘制的伯德图曲线，分析该RC无源低通滤波器的频率特性。
	
	\subsection{理论分析与仿真}
		\subsubsection{第一问}
			图\ref{fig:一阶RC无源低通滤波器原理图}是基本的一阶RC无源低通滤波器原理图。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{一阶RC无源低通滤波器原理图}
				\caption{一阶RC无源低通滤波器原理图}
				\label{fig:一阶RC无源低通滤波器原理图}
			\end{figure}\par
			根据电容的复频域阻抗$Z_C=\frac{1}{Cs}$和电路的串联分压公式得到:
			\begin{gather}
				u_o=\frac{\frac{1}{Cs}}{R+\frac{1}{Cs}}u_i=\frac{1}{RCs+1}u_i
			\end{gather}\par
			稍作变换得到该一阶RC无源低通滤波器的传递函数：
			\begin{gather}
				G(s)=\frac{u_o}{u_i}=\frac{1}{RCs+1}
			\end{gather}\par
			令$s=j\omega$，可以得到一阶RC无源低通滤波器的幅频函数和相频函数：
			\begin{gather}
				H(j\omega)=\frac{1}{jRC\omega+1}  \label{equ:一阶RC无源低通滤波器频域函数}
			\end{gather}\par
			将式\ref{equ:一阶RC无源低通滤波器频域函数}表示成归一化形式：
			\begin{gather}
				H(j\omega)=H_0\frac{1}{1+j\omega/\omega_0}
			\end{gather}\par
			其中，归一化频率$\omega_0=\frac{1}{RC}$，截止频率$f$满足等式：
			\begin{gather}
				f=\frac{1}{2\pi \omega_0}=\frac{1}{2\pi RC}  \label{equ:一阶RC无源低通滤波器的截止频率}
			\end{gather}\par
			以上理论完备，在Multisim软件中进行仿真实验。\par
			由于需要滤除的高频信号为5kHz，需要保留的低频信号为2kHz。所以，设置截止频率为$f_0=4$kHz，设置电容容值为$C=1$nF(Standeard Value,$5\%$)\par
			根据式\ref{equ:一阶RC无源低通滤波器的截止频率}，电阻阻值$R=\frac{1}{2\pi Cf}=39.788$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R=40.2$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			将要求的信号接入该电路，得到仿真电路图，如图\ref{fig:一阶RC无源低通滤波器仿真原理图}所示。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{./7.第七题示波器现象/一阶RC无源低通滤波器仿真原理图}
				\caption{一阶RC无源低通滤波器仿真原理图}
				\label{fig:一阶RC无源低通滤波器仿真原理图}
			\end{figure}\par
			对输入和输出信号做傅里叶变换，得到对应频谱曲线，如图\ref{fig:仿真一阶RC无源低通滤波器频谱曲线}所示。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{./7.第七题示波器现象/仿真一阶RC无源低通滤波器频谱曲线}
				\caption{仿真一阶RC无源低通滤波器频谱曲线}
				\label{fig:仿真一阶RC无源低通滤波器频谱曲线}
			\end{figure}\par
			对于低频信号（2kHz正弦信号），幅值从0.399455降到了0.356448，可求得增益为$A=-20\lg(\frac{0.399455}{0.356448})=-0.98943$dB；对于高频信号（5kHz正弦信号），幅值从0.0594916降到了0.0367441，可求得增益为$A=-20\lg(\frac{0.0594916}{0.0367441})=-4.18536$dB。\par
			这说明图\ref{fig:一阶RC无源低通滤波器原理图}的低通滤波器可以对高频信号进行滤波操作。

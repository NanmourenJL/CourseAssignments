	\subsection{题目重述}
		\begin{compactenum}
			\item 分别设计抑制题6[1]e)中的高、低频信号的滤波器电路，用Multisim对滤波器的频率特性进行仿真。
			\item 搭建实测电路，分别为上述低通实测电路引入一个 20mV 正弦信号，调整信号频率，在 0.1Hz-1MHz 范围内测试 30 个以上关键频率点的输出波形，记录输出信号幅值变化，用 MATLAB 描点绘制 Bode 图，用术语描述滤波器的频率特性。
			\item 做无源RC、一阶有源RC、Sallen-Key结构二阶低通滤波器的对比Bode图，叙述差异。
		\end{compactenum}
	\subsection{题目分析与实施方案}
		\subsubsection{第一问}
			第一问要求设计二阶有源低通滤波器、二阶有源高通滤波器，二阶有源低通滤波器一般由两个独立的储能电容、外围电阻和一个运算放大器组成，而将二阶有源低通滤波器电路中起滤波作用的电阻、电容互换，即可变成二阶有源高通滤波器，因此考虑使用这个电路。\par
			电路确定后，通过计算无源网络的传递函数，得到对应相频特性函数。将需要滤除的频率代入特性曲线的截止频率，便可以推算出对应电路参数，并根据实际情况，选用相关精度的标准阻值电阻。 \par
			然后，使用 Multisim 软件仿真对应电路，将题 6[1]e) 中的输出信号输入搭建的二阶有源低通滤波器、二阶有源高通滤波器上，并连接示波器。
		\subsubsection{第二问}
			根据第一问搭建的电路，向二阶有源低通滤波器等对数频率，分别测试输出信号的幅值变化三十组。\par 
			使用 MATLAB 的 plot 函数，描点绘制出测试得到的参数，以得到相频伯德图曲线。\par 
			根据绘制的伯德图曲线，分析该二阶有源低通滤波器的频率特性。
		\subsubsection{第三问}
			根据本题Sallen-Key结构二阶低通滤波器和之前得到无源RC滤波器、一阶有源RC滤波器的伯德图和结论，相互对比，进行总结归纳。
	
	
	\subsection{理论分析与仿真}
		\subsubsection{第一问}
			\paragraph{\qquad\quad Sallen-Key结构二阶有源低通滤波器}\par
				图\ref{fig:Sallen-Key结构二阶有源低通滤波器电路原理图}是一个Sallen-Key结构二阶有源低通滤波器电路。
				\begin{figure}[h]
					\centering
					\includegraphics[width=0.45\textwidth]{Sallen-Key结构二阶有源低通滤波器电路}
					\caption{Sallen-Key结构二阶有源低通滤波器电路原理图}
					\label{fig:Sallen-Key结构二阶有源低通滤波器电路原理图}
				\end{figure}\par
				根据电路计算可得：
				\begin{align}
					\begin{dcases}
						v_P=v_N\\
						v_N=\frac{R_A}{R_A+R_B}v_o\\
						\frac{v_i-V_a}{R_1}+\frac{v_o-v_a}{\frac{1}{C_1s}}+\frac{v_P-v_a}{R_2}=0\\
						\frac{v_a-v_P}{R_2}=\frac{v_P-0}{\frac{1}{C_2s}}
					\end{dcases}
				\end{align}\par
				整理后，可得Sallen-Key结构二阶有源低通滤波器电路的传递函数为：
				\begin{gather}
					G(s)=\frac{v_o}{v_i}=(1+\frac{R_B}{R_A})\frac{1}{R_1C_1R_2C_2s^2+(R_1C_2+R_2C_2-\frac{R_B}{R_A}R_1C_1)s+1}
					\label{eq:Sallen-Key结构二阶有源低通滤波器传递函数}
				\end{gather}\par
				令$s=j\omega$，并将其表示为归一化形式，可得：
				\begin{align}
					G(j\omega)&=(1+\frac{R_B}{R_A})\frac{1}{1-\omega^2R_1C_1R_2C_2+j\omega(R_1C_2+R_2C_2-\frac{R_B}{R_A}R_1C_1)}\\
					&=H_0\frac{1}{1-(\omega/\omega_0)^2+(j\omega/\omega_0)/Q}
				\end{align}\par
				因此归一化频率$\omega_0$、归一化增益$H_0$和归一化品质因数$Q$，满足：
				\begin{gather}
					\begin{dcases}
						H_0=1+\frac{R_B}{R_A}	\\
						\omega_0 =\frac{1}{\sqrt{R_1C_1R_2C_2}}		\\
						Q =	\frac{1}{-\frac{R_B}{R_A}\sqrt{\frac{R_1C_1}{R_2C_2}}+\sqrt{\frac{R_1C_2}{R_2C_1}}+\sqrt{\frac{R_2C_2}{R_1C_1}}}
					\end{dcases}
				\end{gather}\par
				为了方便参数计算，以下采用等值元件法简化。令$R_1=R_2=R$、$C_1=C_2=C$，则归一化方程组可化为：
				\begin{gather}
					\begin{dcases}
						H_0=1+\frac{R_B}{R_A}	\\
						\omega_0 =\frac{1}{RC}		\\
						Q =	\frac{R_A}{2R_A-R_B}
					\end{dcases}
					\label{eq:Sallen-Key结构二阶有源低通滤波器截止频率}
				\end{gather}\par
				以上理论完备，在Multisim软件中进行仿真实验。\par
				由于需要滤除的高频信号为1kHz，需要保留的中频信号为500Hz。因此，设置截止频率为$f_0=750$Hz。\par
				设置电容容值为$C=1$nF(Standeard Value,$5\%$)\par
				根据式\ref{eq:Sallen-Key结构二阶有源低通滤波器截止频率}，电阻阻值$R=212.2065908$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R=210$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
				同时，我们希望使用同相放大器对中频信号进行补偿，即当$f_M=500$Hz时，增益为1：
				\begin{gather}
					A(\omega_M)=\lvert G(j\omega_M) \rvert =1
				\end{gather}\par
				将相关参数代入计算，整理后得到：
				\begin{gather}
					R_B=0.1823223654R_A
				\end{gather}\par
				设置电阻$R_A$的阻值为$R_A=110$k$\Omega$(Standeard Value,$1\%$)，则$R_B$阻值应为$R_B=20.05545987$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R_B=20$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
				将要求的信号接入该电路，得到仿真电路图，如图\ref{fig:Sallen-Key结构二阶有源低通滤波器仿真原理图}所示。
				\begin{figure}[H]
					\centering
					\includegraphics[width=0.45\textwidth]{./10.第十题示波器现象/10-1仿真原理图}
					\caption{Sallen-Key结构二阶有源低通滤波器仿真原理图}
					\label{fig:Sallen-Key结构二阶有源低通滤波器仿真原理图}
				\end{figure}\par
				对输入和输出信号做傅里叶变换，得到对应频谱曲线，如图\ref{fig:仿真Sallen-Key结构二阶有源低通滤波器频谱曲线}所示。
				\begin{figure}[h]
					\centering
					\includegraphics[width=0.45\textwidth]{./10.第十题示波器现象/10-1仿真频率曲线}
					\caption{仿真Sallen-Key结构二阶有源低通滤波器频谱曲线}
					\label{fig:仿真Sallen-Key结构二阶有源低通滤波器频谱曲线}
				\end{figure}\par
				对于低频信号（200Hz正弦信号），幅值从0.399981升到了0.451616，可求得增益为$A=20\lg(\frac{0.451616}{0.399981})=1.05459916$dB；对于中频信号（500Hz正弦信号），幅值从0.399934降到了0.356566，可求得增益为$A=20\lg(\frac{0.356566}{0.399934})=-0.9969679547$dB；对于高频信号（1kHz正弦信号），幅值从0.0999416降到了0.0470023，可求得增益为$A=20\lg(\frac{0.0470023}{0.0999416})=-6.552543756$dB。\par
				这说明图\ref{fig:Sallen-Key结构二阶有源低通滤波器电路原理图}的低通滤波器可以对高频信号进行滤波操作。
				
			\paragraph{\qquad\quad Sallen-Key结构二阶有源高通滤波器}\par
				图\ref{fig:Sallen-Key结构二阶有源高通滤波器电路原理图}是一个基本的Sallen-Key结构二阶有源高通滤波器电路。
				\begin{figure}[h]
					\centering
					\includegraphics[width=0.45\textwidth]{Sallen-Key结构二阶有源高通滤波器电路}
					\caption{Sallen-Key结构二阶有源高通滤波器电路原理图}
					\label{fig:Sallen-Key结构二阶有源高通滤波器电路原理图}
				\end{figure}\par
				根据电路计算可得：
				\begin{align}
					\begin{dcases}

						v_N=\frac{R_A}{R_A+R_B}v_o\\

					\end{dcases}
				\end{align}\par
				整理后，可得Sallen-Key结构二阶有源高通滤波器的传递函数为：

					G(s)=\frac{v_o}{v_i}=(1+\frac{R_B}{R_A})\frac{R_1C_1R_2C_2s^2}{R_1C_1R_2C_2s^2+(R_1C_2+R_2C_2-\frac{R_B}{R_A}R_1C_1)s+1}
				\end{gather}\par
				令$s=j\omega$，并将其表示为归一化形式，可得：

		\subsubsection{结论}
		10.1 经过验证，所设计二阶有源高通低通滤波器均可正常工作，仿真结果与预期结果误差再允许范围之内，由此判断所设计电路正确。\par
		10.2 滤波器的频率特性：通过搭建二阶有源低通滤波器实测电路并进行测试，当输入 20mV 正弦信号且在 0.1Hz 至 1MHz 范围内改变频率测试多个关键频率点后，由 MATLAB 绘制的 Bode 图可知，在低频段输出信号幅值较为稳定且接近通带增益对应的幅值，随着频率升高，输出幅值逐渐下降，在截止频率处，输出幅值下降至通带幅值的约 0.707 倍，之后高频段快速衰减。体现出二阶有源低通滤波器具有通低频阻高频、二阶有源高通滤波器具有通高频阻低频的典型频率特性，且在截止频率附近呈现出相应的过渡特性，符合理论预期。\par
		10.3 无源RC、一阶有源RC、Sallen-Key结构二阶低通滤波器的差异：无源 RC 低通滤波器通带增益小于 1 且衰减信号，截止特性平缓、过渡带宽，相移随频率渐变；一阶有源 RC 低通滤波器可设大于 1 的通带增益，过渡带较无源 RC 有所改善但仍较缓，相移特性因运放与无源 RC 有别；Sallen - Key 结构二阶低通滤波器通带增益可灵活设定，截止特性陡峭、过渡带窄，对高频滤除效果佳，相移在截止频率附近变化复杂且斜率大。\par

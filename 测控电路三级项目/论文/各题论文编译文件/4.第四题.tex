	\subsection{题目重述}
		\begin{compactenum}
			\item 选用适当参数，完成$-20$倍反相放大电路的Multisim仿真，进行仿真测试。
			\item 焊装电路，做如下测试，并分别叙述与相同输入下仿真波形的差异。
			\item 选用适当参数，利用T型反馈网络实现-300或-400倍反相放大电路的Multisim仿真。
			\item 通过拆换元件，搭建$-300$（约）或$-400倍$（约）T型反馈网络反相高增益放大电路。
		\end{compactenum}
	\subsection{题目分析与实施方案}
		\subsubsection{第一问}
			第一问需要构建一个反相放大电路，并且实现$-20$倍的放大倍数。然后使用Multisim对这个电路在不同输入信号下的放大效果进行测试，输入信号包括不同频率的正弦波和特定占空比的矩形波，通过仿真来观察输出信号的特性。
		\subsubsection{第二问}
			第二问依据之前设计的$-20$倍反相放大电路原理图，在PCB上进行电路焊装。并对焊装后的$-20$倍反相放大电路进行多方面测试，并对比与仿真波形的差异。对不同峰值和频率的正弦波、矩形波、三角波信号放大测试，能全面了解电路在实际焊装后的性能表现，包括频率响应、压摆率限制以及阶跃响应等特性，进而分析实际电路与理想仿真情况的偏差原因。
		\subsubsection{第三问}
			第三问利用反相放大器改装T型反馈网络，要求按照T型反馈网络的结构图搭建符合题目要求设计一个$-300$倍放大的反相高增益放大电路，并进行正弦信号的Multsim仿真验证。
		\subsubsection{第四问}
			第四问需要按照上一问设计的$-300$倍放大的反相高增益放大电路，选用合适的电阻，在PCB上进行电路焊装。并对焊装后的电路进行相应信号测试。并适当调整信号幅度，得到无失真放大的极限条件。
	
	\subsection{理论分析与仿真}
		\subsubsection{第一问}
			图\ref{fig:反相放大电路}是一个基本的反相放大电路。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{反相放大电路}
				\caption{反相放大电路}
				\label{fig:反相放大电路}
			\end{figure}\par
			由于差分输入电阻$r_{id}$较大，一般认为同相和反相输入端直接没有电流流经，直接忽略该电流。于是，可依据电路结构，写出同相输入端$v_P$和反相输入端$v_N$的电压公式
			\begin{align}
				v_P&=0 \\
				\frac{v_N-v_I}{R_1}&=\frac{v_O-v_N}{R_2}
			\end{align}\par
			结合等式$u_O=A_{od}u_I=A_{od}(v_P-v_N)$，联立可得到
			\begin{align}
				v_O=A_{od}(-\frac{1}{1+R_1/R_2}v_I-\frac{1}{1+R_1/R_2}v_O)
			\end{align}\par
			定义实际放大倍数$A'$为输出电压$v_O$和输入电压$v_I$之比，则有
			\begin{align}\label{eq:实际反相放大倍数}
				A'=\frac{v_O}{v_I}=(-\frac{R_2}{R_1})\frac{1}{1+(1+R_2/R_1)/A_{od}}
			\end{align}\par
			这个结果指出，图\ref{fig:反相放大电路}构成的电路同样是一个放大器，其放大倍数为$A'$。因为$A'$为负，所以$v_O$和$v_I$的极性相反，故而命名为反相放大器（或倒相放大器）。\par
			在式\ref{eq:实际反相放大倍数}中令$A_{od}\rightarrow \infty$就得到理想的放大倍数$A'_{ideal}$:
			\begin{align}\label{eq:理想反相放大倍数}
				A'_{ideal}=\lim_{A_{od}\rightarrow \infty}A'=-\frac{R_2}{R_1}
			\end{align}\par
			根据理想放大倍数计算公式式\ref{eq:理想反相放大倍数}，为了得到-20倍同相放大电路，只需要满足等式$R_2=20R_1$即可。\par
			以上理论完备，在Multisim软件中进行仿真实验。\par
			根据图\ref{fig:反相放大电路}，搭建仿真电路图，仿真电路使用OP07芯片作为运放芯片，采用$\pm15$V正负电压作为电源供电，反馈电阻网络选择$R_1=1$k$\Omega$，$R_2=20$k$\Omega$电阻将信号发生器接入同相放大电路输入端，示波器通道A接入同相放大电路输入端，示波器通道B接入同相放大电路输出端，其搭建示意图如图\ref{fig:反相放大电路仿真原理图}所示。\par

		\subsubsection{第三问}
			图\ref{fig:T型反馈反相放大电路}是一个T型反馈网络的反相放大电路。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{T型反馈反相放大电路}
				\caption{T型反馈反相放大电路}
				\label{fig:T型反馈反相放大电路}
			\end{figure}\par
			由于差分输入电阻$r_{id}$较大，一般认为同相和反相输入端直接没有电流流经，直接忽略该电流。于是，可依据电路结构，写出同相输入端$v_P$和反相输入端$v_N$的电压公式
			\begin{align}
				v_P&=0 \\
				\frac{v_I-v_N}{R_1}&=\frac{v_N-v_B}{R_2}\\
				\frac{v_N-v_B}{R_2}+\frac{0-v_B}{R_3}&=\frac{v_B-v_O}{R_4}
			\end{align}\par
			结合等式$u_O=A_{od}u_I=A_{od}(v_P-v_N)$，联立可得到
			\begin{align}
				v_O=A_{od}(-\frac{R_2^2R_3+R_2^2R_4+R_2R_3R_4}{(R_1+R_2)(R_2R_3+R_3R_4+R_2R_4)-R_1R_3R_4}v_I-\frac{R_1R_2R_3}{(R_1+R_2)(R_2R_3+R_3R_4+R_2R_4)-R_1R_3R_4}v_O)
			\end{align}\par
			定义实际放大倍数$A"$为输出电压$v_O$和输入电压$v_I$之比，则有
			\begin{align}\label{eq:实际T型反相放大倍数}
				A"=\frac{v_O}{v_I}=-\frac{A_{od}(R_2R_3+R_2R_4+R_3R_4)}{1+A_{od}R_1R_3}
			\end{align}\par
			这个结果指出，图\ref{fig:T型反馈反相放大电路}构成的电路是一个放大器，其放大倍数为$A"$。因为$A"$为负，所以$v_O$和$v_I$的极性相反。\par
			在式\ref{eq:实际T型反相放大倍数}中令$A_{od}\rightarrow \infty$就得到理想的放大倍数$A"_{ideal}$:
			\begin{align}\label{eq:理想T型反馈反相放大倍数}
				A"_{ideal}=\lim_{A_{od}\rightarrow \infty}A"=-\frac{R_2R_3+R_2R_4+R_3R_4}{R_1R_3}
			\end{align}\par
			根据理想放大倍数计算公式式\ref{eq:理想T型反馈反相放大倍数}，为了得到-300倍同相放大电路，只需要满足等式$R_2=20R_1$即可。\par
			以上理论完备，在Multisim软件中进行仿真实验。\par
			根据图\ref{fig:反相放大电路}，搭建仿真电路图，仿真电路使用OP07芯片作为运放芯片，采用$\pm15$V正负电压作为电源供电，反馈电阻网络选择$R_1=1$k$\Omega$，$R_2=R_3=R_4=100$k$\Omega$电阻，将信号发生器接入同相放大电路输入端，示波器通道A接入同相放大电路输入端，示波器通道B接入同相放大电路输出端，其搭建示意图如图\ref{fig:T型反馈反相放大电路仿真原理图}所示。\par

				\centering
				\includegraphics[width=0.4\textwidth]{./4.第四题示波器现象/T型反馈反相放大电路仿真原理图}
				\caption{T型反馈反相放大电路仿真原理图}
				\label{fig:T型反馈反相放大电路仿真原理图}
			\end{figure}\par
			该仿真电路图测得对应的仿真正弦波形如图\ref{fig:仿真T型反馈反相放大电路放大正弦信号}所示。

	\subsection{实际电路测试结果与数据分析}
		\subsubsection{第二问}
			由于PCB给出的反相放大电路是根据T型反馈网络反相放大电路设计的，因此在实际使用时需将$R_{5\_1}$短路，$R_{5\_2}$开路，来形成反馈网络。此时根据等式$R_2=20R_1$，若其中$R_1$选用阻值为$=1.1$k$\Omega$的电阻，则可以得到$R_2$的阻值为$R_1=22$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R=22$k$\Omega$(Standeard Value,$1\%$)的电阻。
			实际电路板除了以上$R_1$、$R_2$电阻以外还存在输入平衡电阻$R_3$，该电阻是为了维持运算放大器输入级差分电路的对称性，因此该电阻等于负输入端的对地等效电阻，即$R_3=\frac{1}{\frac{1}{R_1}+\frac{1}{R_2}}=\frac{R_1R_2}{R_1+R_2}=\frac{22}{21}$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R=1.05$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			焊接后，经检查分立元件位置端正、放置平整、焊点表面光滑、无凸起毛刺、无虚焊，周围无助焊剂残留；多引脚器件无短路，焊点均匀一致。符合测试要求。\par

			图\ref{fig:实际反相-20倍放大器放大20mV、100Hz、80方波信号}和图\ref{fig:仿真反相-20倍放大器放大20mV、100Hz、占空比80方波信号}的图像基本一致。将图像放大来看，信号在上升沿、下降沿处是个阶跃响应，运放芯片会随着信号幅值的改变产生一定的量的超调。幅值越大，超调量越大。同时也调试发现，随着频率信号的升高，阶跃响应的上升时间会加长，这是因为对于同一芯片的增益带宽积是一定的，随着信号频率的增大，运放的压摆率是逐渐减小的，使得上升时间的延长。同时，对比图\ref{fig:实际反相-20倍放大器放大20mV、100Hz、80方波信号}、图\ref{fig:实际反相-20倍放大器放大20mV、10Hz、80方波信号}和图\ref{fig:实际反相-20倍放大器放大20mV、10kHz、80方波信号}可以发现，改变输入方波信号的频率，会影响阶跃信号的跟随放大效果。观察除法算子算出的信号，低频的10Hz和100Hz对应上升信号仅有短时间的波动，但是高频的10kHz上可以观察到明显的上升曲线，导致了，信号的失真。\par
		\subsubsection{第四问}
			根据式\ref{eq:理想T型反馈反相放大倍数}，希望$A"_{ideal}=-300$，对等式求解可以简单获得一个可行解：$R_1=1$k$\Omega$、$R_2=100$k$\Omega$、$R_3=100$k$\Omega$、$R_4=100$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R_1=1$k$\Omega$、$R_2=100$k$\Omega$、$R_3=100$k$\Omega$、$R_4=100$k$\Omega$(Standeard Value,$1\%$)的电阻。
			实际电路板除了以上$R_1$、$R_2$电阻以外还存在输入平衡电阻$R_3$，该电阻是为了维持运算放大器输入级差分电路的对称性，因此该电阻等于负输入端的对地等效电阻，即$R=R_1//(R_2+R_3//R_4)=\frac{151}{150}$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R=1.00$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			焊接后，经检查分立元件位置端正、放置平整、焊点表面光滑、无凸起毛刺、无虚焊，周围无助焊剂残留；多引脚器件无短路，焊点均匀一致。符合测试要求。\par

	\subsection{结论与收获}
		\subsubsection{结论}
			\paragraph{\qquad -20 倍反相放大电路 Multisim 仿真结论}
			对于 1KHz、峰值 20mV 正弦信号，能实现 -20 倍放大，输出波形与输入波形形状相同，相位相反，放大倍数准确。\par
			1MHz 正弦信号放大时，输出波形仍为正弦，但幅值略小于理论值，因运放带宽限制，高频信号有一定衰减。
			对峰值 2V、100KHz 正弦信号放大，输出信号失真，未达到 -20 倍放大效果。\par
			对峰值 20mV、占空比 80\%、频率 100Hz 矩形波信号放大，输出矩形波反相，形状基本保持，仅在边沿有微小变化。
			\paragraph{\qquad 焊装 -20 倍反相放大电路测试结论}
			内部失调调零可有效减小输出失调电压，使直流输出更接近 0V。\par
			对 200mV 直流信号放大，能实现 -20 倍放大，输出为 -4V，但有微小偏差，与元件实际精度有关。
			1KHz 和 1MHz 正弦信号放大，1KHz 时与仿真结果相似，1MHz 时输出波形顶部变圆钝，因实际运放高频性能比仿真模型差，由于摆率限制，输出波形顶部和底部出现削波失真。\par
			对峰值 2V、100KHz 正弦信号放大，对矩形波信号放大，阶跃响应上升沿和下降沿有倾斜，原因是运放带宽有限以及电路中存在寄生电容和电感等分布参数，影响信号快速变化时的响应。
			\paragraph{\qquad -300（约）T 型反馈网络反相放大电路 Multisim 仿真结论}
			对 5mV、100Hz 正弦信号放大，可实现约 -300 或 -400 倍放大，输出波形正弦但是略有失真。
			\paragraph{\qquad 搭建 -300（约）T 型反馈网络反相高增益放大电路测试结论}
			仿真与实测对 5mV、100Hz 正弦信号放大倍数有一定差异，实测因元件实际值与标称值偏差以及焊接等因素导致放大倍数不准确，且输出波形有更多噪声。
		\subsubsection{小组成员收获}
			\paragraph{\qquad 理论分析层面}
			深入理解了反相放大电路的工作原理，包括普通反相放大电路和 T 型反馈网络反相放大电路的增益计算方法，明白了电路中各元件参数对放大倍数、带宽、输入输出阻抗等性能指标的影响。通过分析信号频率、幅值以及运放摆率等因素对输出波形的影响，掌握了电路频率响应和非线性失真的相关理论知识，如带宽限制导致高频信号放大倍数下降、摆率不足引起大信号或高频信号失真等。对阶跃响应的研究，让成员们学习了电路动态特性的分析方法，了解了影响阶跃响应的多种因素，如运放的压摆率、带宽、反馈网络参数以及电路中的寄生参数等。

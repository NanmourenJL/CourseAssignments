	\subsection{题目重述}
		\begin{compactenum}
			\item 设计抑制题6[1]e)中的中频信号的滤波器电路，用Multisim对滤波器的频率特性进行仿真。
			\item 搭建实测电路，分别为上述低通实测电路引入一个 20mV 正弦信号，调整信号频率，在 0.1Hz-1MHz 范围内测试 30 个以上关键频率点的输出波形，记录输出信号幅值变化，用 MATLAB 描点绘制 Bode 图，用术语描述滤波器的频率特性。
			\item 说明所搭建的实际电路是否能够完成任务。
		\end{compactenum}
	\subsection{题目分析与实施方案}
		\subsubsection{第一问}
			第一问要求设计无源带通滤波器，无源带通滤波器一般由两个电容和两个电阻组成，考虑使用这个电路。\par
			电路确定后，通过计算无源网络的传递函数，得到对应相频特性函数。将需要滤除的频率代入特性曲线的截止频率，便可以推算出对应电路参数，并根据实际情况，选用相关精度的标准阻值电阻。 \par
			然后，使用 Multisim 软件仿真对应电路，将题 6[1]e) 中的输出信号输入搭建的无源带通滤波器上，并连接示波器。
		\subsubsection{第二问}
			根据第一问搭建的电路，向无源带通滤波器等对数频率，分别测试输出信号的幅值变化三十组。\par 
			使用 MATLAB 的 plot 函数，描点绘制出测试得到的参数，以得到相频伯德图曲线。\par 
			根据绘制的伯德图曲线，分析该无源带通滤波器的频率特性。
		\subsubsection{第三问}
			根据第二问的伯德图，测定需过滤处信号和需保留信号处的增益。判断截止频率是否在合理范围以内，以评价电路的滤波效果。
	
	\subsection{理论分析与仿真}
		\subsubsection{第一问}
			图\ref{fig:无源带通滤波器电路原理图}是一个无源带通滤波器电路。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{无源带通滤波器电路原理图}
				\caption{无源带通滤波器电路原理图}
				\label{fig:无源带通滤波器电路原理图}
			\end{figure}\par
			根据电路计算可得：
			\begin{align}
				\begin{dcases}
					\frac{v_a-v_o}{\frac{1}{C_2s}}=\frac{v_o}{R_2}	\\
					\frac{v_i-v_a}{R_1}=\frac{v_a}{\frac{1}{C_1s}}+\frac{v_o}{R_2}	\\
				\end{dcases}
			\end{align}\par
			整理后，可得无源带通滤波器电路的传递函数为：
			\begin{gather}
				G(s)=\frac{v_o}{v_i}=\frac{R_2C_2s}{R_1C_1R_2C_2s^2+(R_1C_2+R_2C_2+R_1C_1)s+1}
			\end{gather}\par
			令$s=j\omega$，并将其表示为归一化形式，可得：
			\begin{align}
				G(j\omega)&=\frac{j\omega R_2C_2}{1-\omega^2R_1C_1R_2C_2+j\omega(R_1C_2+R_2C_2+R_1C_1)}\\
				&=H_0\frac{(j\omega/\omega_0)/Q}{1-(\omega/\omega_0)^2+(j\omega/\omega_0)/Q}
			\end{align}\par
			因此归一化频率$\omega_0$、归一化增益$H_0$和归一化品质因数$Q$，满足：
			\begin{gather}
				\begin{dcases}
					H_0=\frac{R_1C_2+R_2C_2+R_1C_1}{R_2C_2}	\\
					\omega_0 =\frac{1}{\sqrt{R_1C_1R_2C_2}}		\\
					Q =	\frac{\sqrt{R_1C_1R_2C_2}}{R_1C_2+R_2C_2+R_1C_1}
				\end{dcases}
			\end{gather}\par
			为了方便参数计算，以下采用等值元件法简化。令$C_1=C_2=C$，则归一化方程组可化为：
			\begin{gather}
				\begin{dcases}
					H_0=\frac{2R_1+R_2}{R_2}	\\
					\omega_0 =\frac{1}{\sqrt{R_1R_2}C}		\\
					Q =	\frac{\sqrt{R_1R_2}}{2R_1+R_2}
				\end{dcases}
				\label{eq:无源带通滤波器截止频率}
			\end{gather}\par
			以上理论完备，在Multisim软件中进行仿真实验。\par
			由于需要滤除的低频信号为200Hz，高频信号为1kHz，需要保留的中频信号为500Hz。因此，设置中心频率为$f_0=500$Hz，归一化品质因数$Q=0.35$。\par
			设置电容容值为$C=100$nF(Standeard Value,$5\%$)\par
			根据式\ref{eq:无源带通滤波器截止频率}，电阻阻值$R_1=1952.1005026853\Omega$、$R_2=5190.3671713092\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R_1=1.96$k$\Omega$(Standeard Value,$1\%$)、$R_2=5.23$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			将要求的信号接入该电路，得到仿真电路图，如图\ref{fig:无源带通滤波器仿真原理图}所示。
			\begin{figure}[H]
				\centering
				\includegraphics[width=0.45\textwidth]{./11.第十一题示波器现象/11仿真原理图}
				\caption{无源带通滤波器仿真原理图}
				\label{fig:无源带通滤波器仿真原理图}
			\end{figure}\par
			对输入和输出信号做傅里叶变换，得到对应频谱曲线，如图\ref{fig:无源带通滤波器频谱曲线}所示。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{./11.第十一题示波器现象/11仿真频率曲线}
				\caption{无源带通滤波器频谱曲线}
				\label{fig:无源带通滤波器频谱曲线}
			\end{figure}\par
			对于低频信号（200Hz正弦信号），幅值从0.399988降到了0.184756，可求得增益为$A=20\lg(\frac{0.184756}{0.399988})=-6.708968223$dB；对于中频信号（500Hz正弦信号），幅值从0.399956降到了0.228607，可求得增益为$A=20\lg(\frac{0.228607}{0.399956})=-4.588453837$dB；对于高频信号（1kHz正弦信号），幅值从0.0999533降到了0.0504747，可求得增益为$A=20\lg(\frac{0.0504747}{0.0999533})=-5.934467815$dB。\par
			这说明图\ref{fig:无源带通滤波器电路原理图}的带通滤波器可以对高频信号和低频信号进行滤波操作。

		\subsubsection{第三问}
			根据表\ref{tab:不同频率下无源带通滤波器增益表}中的数据，可以发现，当频率为500Hz的时候增益最大，此时为0.58，也就是-4.73dB，小于要求的-3dB增益，说明该电路对所有信号，均有抑制作用，没有很好的完成滤波的要求。\par
			因此，可以认为实际所搭建的电路无法完成任务。

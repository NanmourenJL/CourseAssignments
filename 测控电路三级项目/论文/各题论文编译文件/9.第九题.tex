	\subsection{题目重述}
		\begin{compactenum}
			\item 设计一个反相一阶有源低通滤波器，分别构建仿真电路和实测电路，滤除题6[1]d)中的高频信号。
			\item 设计一个反相一阶有源高通滤波器，分别构建仿真电路和实测电路，滤除题6[1]d)中的低频信号。
			\item 搭建实测电路，分别为上述低通实测电路引入一个20mV正弦信号，调整信号频率，在0.1Hz-1MHz 范围内测试30个以上关键频率点的输出波形，记录输出信号幅值变化，用MATLAB描点绘制Bode图，用术语描述滤波器的频率特性源。
			\item 阐述对截止频率选择的见解，并通过实例或者仿真加以说明。
		\end{compactenum}
	\subsection{题目分析与实施方案}
		\subsubsection{第一问}
			第一问要求设计反相一阶有源低通滤波器，最常见的反相一阶有源低通滤波器是在积分运算电路的电容$C$上，并联一个电阻$R_2$,即可得到反相输入的一阶低通滤波器，因此考虑在本题使用这个电路。\par 
			电路确定后，通过计算有源低通网络的传递函数，得到对应相频特性函数。将需要滤除的频率代入特性曲线的截止频率，便可以推算出对应电路参数，并根据实际情况，选用相关精度的标准阻值电阻。\par 
			然后，使用 Multisim 软件仿真对应电路，将题 6[1]d) 中的输出信号输入搭建的反相一阶有源低通滤波器上，并连接示波器。观察示波器波形，判断高频信号是否被滤除。\par
		\subsubsection{第二问}
			第二问要求设计反相一阶有源高通滤波器，根据对偶性，将反相一阶低通滤波电路中的电容换成电阻、电阻换成电容，就可得到反相输入的一阶高通滤波器，因此考虑在本题使用这个电路。 \par
			电路确定后，通过计算有源高通网络的传递函数，得到对应相频特性函数。将需要滤除的频率代入特性曲线的截止频率，便可以推算出对应电路参数，并根据实际情况，选用相关精度的标准阻值电阻。 \par
			然后，使用 Multisim 软件仿真对应电路，将题 6[1]d) 中的输出信号输入搭建的反相一阶有源高通滤波器上，并连接示波器。观察示波器波形，判断高频信号是否被滤除。\par
		\subsubsection{第三问}
			根据第一问搭建的电路，向反相一阶有源低通滤波器等对数频率，分别测试输出信号的幅值变化三十组。\par 
			使用 MATLAB 的 plot 函数，描点绘制出测试得到的参数，以得到相频伯德图曲线。 \par
			根据绘制的伯德图曲线，分析该反相一阶有源低通滤波器的频率特性。
		\subsubsection{第四问}
			第四问需要根据具体产生的实验现象和关于截止频率的相关理论，来深刻阐释截止频率的意义。
			
			
			
			
	\subsection{理论分析与仿真}
		\subsubsection{第一问}
			图\ref{fig:反相一阶有源低通滤波器电路原理图}是反相一阶有源低通滤波器电路原理图。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.6\textwidth]{反相一阶有源低通滤波器}
				\caption{反相一阶有源低通滤波器电路原理图}
				\label{fig:反相一阶有源低通滤波器电路原理图}
			\end{figure}\par
			根据电路计算可得：
			\begin{gather}
				\frac{u_i-0}{R_g}=\frac{0-u_o}{\frac{1}{Cs}//R}
			\end{gather}\par
			由此可得：
			\begin{gather}
				G(s)=\frac{u_o}{u_i}=-\frac{R}{R_g}\frac{1}{RCs+1}
			\end{gather}\par
			令$s=j\omega$，并将其表示为归一化形式，可得：
			\begin{align}
				G(j\omega)&=-\frac{R}{R_g}\frac{1}{j\omega RC+1}\\
				&=H_0\frac{1}{1+j\omega/\omega_0}
			\end{align}\par
			因此归一化频率$\omega_0$和归一化增益$H_0$，满足：
			\begin{gather}
				\begin{cases}
					\omega_0 =\frac{1}{RC}=2\pi f_0
					\\
					H_0=-\frac{R}{R_g}
				\end{cases}
				\label{eq:反相一阶有源低通滤波器截止频率}
			\end{gather}\par
			以上理论完备，在Multisim软件中进行仿真实验。\par
			由于需要滤除的高频信号为5kHz，需要保留的低频信号为2kHz。因此，设置截止频率为$f_0=4$kHz。设置电容容值为$C=1$nF(Standeard Value,$5\%$)\par
			根据式\ref{eq:反相一阶有源低通滤波器截止频率}，电阻阻值$R=\frac{1}{2\pi Cf_0}=39.788$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R=40.2$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			同时，我们希望使用反相放大器对低频信号进行补偿，即当$f_L=2$kHz时，增益为1：
			\begin{gather}
				A(\omega_L)=\lvert G(j\omega_L) \rvert =-\frac{R}{R_g}\frac{1}{\sqrt{1+(RC\omega_L)^2}}
			\end{gather}\par
			将$\omega_L=2\pi f_L=4000\pi$、$R=40.2$k$\Omega$、$C=1$nF代入计算，整理后得到：
			\begin{gather}
				R_g=0.8925744129R
			\end{gather}\par
			其中$R$的阻值已经求得为$40.2$k$\Omega$，则$R_g$阻值应为$R_g=35.8814914$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R_g=35.7$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			将要求的信号接入该电路，得到仿真电路图，如图\ref{fig:反相一阶有源低通滤波器仿真原理图}所示。
			\begin{figure}[H]
				\centering
				\includegraphics[width=0.45\textwidth]{./9.第九题示波器现象/9-1仿真原理图}
				\caption{反相一阶有源低通滤波器仿真原理图}
				\label{fig:反相一阶有源低通滤波器仿真原理图}
			\end{figure}\par
			对输入和输出信号做傅里叶变换，得到对应频谱曲线，如图\ref{fig:仿真反相一阶有源低通滤波器频谱曲线}所示。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{./9.第九题示波器现象/9-1仿真频率曲线}
				\caption{仿真反相一阶有源低通滤波器频谱曲线}
				\label{fig:仿真反相一阶有源低通滤波器频谱曲线}
			\end{figure}\par
			对于低频信号（2kHz正弦信号），幅值从0.399455降到了0.399694，可求得增益为$A=-20\lg(\frac{0.399455}{0.399694})=0.005195345$dB；对于高频信号（5kHz正弦信号），幅值从0.0594916降到了0.0411997，可求得增益为$A=-20\lg(\frac{0.0594916}{0.0411997})=-3.191231911$dB。\par
			这说明图\ref{fig:反相一阶有源低通滤波器电路原理图}的低通滤波器可以对高频信号进行滤波操作。
			
		\subsubsection{第二问}
			图\ref{fig:反相一阶有源高通滤波器电路原理图}是反相一阶有源高通滤波器电路原理图。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{反相一阶有源高通滤波器原理图}
				\caption{反相一阶有源高通滤波器电路原理图}
				\label{fig:反相一阶有源高通滤波器电路原理图}
			\end{figure}\par
			根据电路计算可得：
			\begin{gather}
				\frac{u_i-0}{R//\frac{1}{Cs}}=\frac{0-u_o}{R_f}
			\end{gather}\par
			由此可得：
			\begin{gather}
				G(s)=\frac{u_o}{u_i}=-\frac{R_f}{R}\frac{RCs}{RCs+1}
			\end{gather}\par
			令$s=j\omega$，并将其表示为归一化形式，可得：
			\begin{align}
				G(j\omega)&=-\frac{R_f}{R}\frac{j\omega RC}{j\omega RC+1}\\
				&=H_0\frac{j\omega/\omega_0}{1+j\omega/\omega_0}
			\end{align}\par
			因此归一化频率$\omega_0$和归一化增益$H_0$，满足：
			\begin{gather}
				\begin{cases}
					\omega_0 =\frac{1}{RC}=2\pi f_0
					\\
					H_0=-\frac{R_f}{R}
				\end{cases}
				\label{eq:反相一阶有源高通滤波器截止频率}
			\end{gather}\par
			以上理论完备，在Multisim软件中进行仿真实验。\par
			由于需要滤除的低频信号为2kHz，需要保留的高频信号为5kHz。因此，设置截止频率为$f_0=4$kHz。设置电容容值为$C=1$nF(Standeard Value,$5\%$)\par
			根据式\ref{eq:反相一阶有源高通滤波器截止频率}，电阻阻值$R=\frac{1}{2\pi Cf_0}=39.788$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R=40.2$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			同时，我们希望使用反相放大器对低频信号进行补偿，即当$f_H=5$kHz时，增益为1：
			\begin{gather}
				A(\omega_H)=\lvert G(j\omega_H) \rvert =-\frac{R_f}{R}\frac{RC\omega_H}{\sqrt{1+(RC\omega_H)^2}}
			\end{gather}\par
			将$\omega_H=2\pi f_H=10000\pi$、$R=40.2$k$\Omega$、$C=1$nF代入计算，整理后得到：
			\begin{gather}
				R_f=1.275528127R
			\end{gather}\par
			其中$R$的阻值已经求得为$40.2$k$\Omega$，则$R_f$阻值应为$R_f=51.27623$k$\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R_f=51.1$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			将要求的信号接入该电路，得到仿真电路图，如图\ref{fig:反相一阶有源高通滤波器仿真原理图}所示。
			\begin{figure}[H]

				\label{fig:反相一阶有源高通滤波器仿真原理图}
			\end{figure}\par
			对输入和输出信号做傅里叶变换，得到对应频谱曲线，如图\ref{fig:仿真反相一阶有源高通滤波器频谱曲线}所示。

		\subsubsection{第四问}
			截止频率是指信号增益开始显著下降的频率点。在滤波器设计中，截止频率决定了哪些频率成分能够通过，哪些则被抑制。因此，选择合适的截止频率对于确保系统性能至关重要。\par
			信号频率的选取需要根据保留信号频率和滤除信号频率决定。一般是选择两者信号的算术平均值或者几何平均值，亦或是该平均值附近的整数值，以此方便计算。

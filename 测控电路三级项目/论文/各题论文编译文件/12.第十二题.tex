	\subsection{题目重述}
		\begin{compactenum}
			\item 设计抑制题6[1]e)中的中频信号的滤波器电路，用Multisim对滤波器的频率特性进行仿真。
			\item 搭建实测电路，分别为上述低通实测电路引入一个 20mV 正弦信号，调整信号频率，在 0.1Hz-1MHz 范围内测试 30 个以上关键频率点的输出波形，记录输出信号幅值变化，用 MATLAB 描点绘制 Bode 图，用术语描述滤波器的频率特性。
		\end{compactenum}
	\subsection{题目分析与实施方案}
		\subsubsection{第一问}
			第一问要求设计一阶有源高通-低通滤波器构成的二阶有源带通滤波器，由一个一阶有源高通和低通滤波器构成，因此考虑在本题使用这个电路。 \par
			电路确定后，通过计算无源网络的传递函数，得到对应相频特性函数。将需要滤除的频率代入特性曲线的截止频率，便可以推算出对应电路参数，并根据实际情况，选用相关精度的标准阻值电阻。\par 
			然后，使用 Multisim 软件仿真对应电路，将题 6[1]d)中的输出信号输入搭建的一阶有源高通-低通滤波器构成的二阶有源带通滤波器上，并连接示波器。观察示波器波形，判断高频信号是否被滤除。
		\subsubsection{第二问}
			根据第一问搭建的电路，向二阶有源带通滤波器等对数频率，分别测试输出信号的幅值变化三十组。\par 
			使用 MATLAB 的 plot 函数，描点绘制出测试得到的参数，以得到相频伯德图曲线。\par 
			根据绘制的伯德图曲线，分析该二阶有源带通滤波器的频率特性。
	
	\subsection{理论分析与仿真}
		\subsubsection{第一问}
			图\ref{fig:二阶有源带通滤波器电路原理图}是一个由一阶同相有源高通滤波器和一个同相有源低通滤波器组成的二阶有源带通滤波器电路。
			\begin{figure}[h]
				\centering
				\includegraphics[width=0.45\textwidth]{二阶有源带通滤波器电路原理图}
				\caption{二阶有源带通滤波器电路原理图}
				\label{fig:二阶有源带通滤波器电路原理图}
			\end{figure}\par
			由于使用两个滤波器级联得到，因此最终的传递函数可以由两个一阶有源滤波器的传递函数相乘得到。因此将式\ref{eq:同相一阶有源低通滤波器传递函数}和式\ref{eq:同相一阶有源高通滤波器传递函数}相乘，得：
			\begin{gather}
				G(s)=(1+\frac{R_4}{R_3})(1+\frac{R_6}{R_5})\frac{R_2C_2s}{R_1C_1R_2C_2s^2+(R_1C_1+R_2C_2)s+1}
			\end{gather}
			令$s=j\omega$，并将其表示为归一化形式，可得：
			\begin{align}
				G(j\omega)&=(1+\frac{R_4}{R_3})(1+\frac{R_6}{R_5})\frac{j\omega R_2C_2}{1-\omega^2R_1C_1R_2C_2+j\omega(R_1C_1+R_2C_2)}\\
				&=H_0\frac{(j\omega/\omega_0)/Q}{1-(\omega/\omega_0)^2+(j\omega/\omega_0)/Q}
			\end{align}\par
			因此归一化频率$\omega_0$、归一化增益$H_0$和归一化品质因数$Q$，满足：
			\begin{gather}
				\begin{dcases}
					H_0=(1+\frac{R_4}{R_3})(1+\frac{R_6}{R_5})\frac{R_2C_2}{R_1C_1+R_2C_2}	\\
					\omega_0 =\frac{1}{\sqrt{R_1C_1R_2C_2}}		\\
					Q =	\frac{\sqrt{R_1C_1R_2C_2}}{R_1C_1+R_2C_2}
				\end{dcases}
			\end{gather}\par
			为了方便参数计算，以下采用等值元件法简化。令$C_1=C_2=C$，则归一化方程组可化为：
			\begin{gather}
				\begin{dcases}
					H_0=(1+\frac{R_4}{R_3})(1+\frac{R_6}{R_5})\frac{R_2}{R_1+R_2}	\\
					\omega_0 =\frac{1}{\sqrt{R_1R_2}C}		\\
					Q =	\frac{\sqrt{R_1R_2}}{R_1+R_2}
				\end{dcases}
				\label{eq:二阶有源带通滤波器截止频率}
			\end{gather}\par
			以上理论完备，在Multisim软件中进行仿真实验。\par
			由于需要滤除的低频信号为200Hz，高频信号为1kHz，需要保留的中频信号为500Hz。因此，设置中心频率为$f_0=500$Hz、归一化品质因数$Q=\frac{\sqrt{2}}{2}$、归一化增益$H_0=1$。\par
			设置电容容值为$C=100$nF(Standeard Value,$5\%$)\par
			根据式\ref{eq:二阶有源带通滤波器截止频率}，电阻阻值$R_1=R_2=3183.098862\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R_1=R_2=3.16$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			将计算得到的电阻代入得$(1+\frac{R_4}{R_3})(1+\frac{R_6}{R_5})=2$。为了方便电阻选取，令$R_3=R_5=R_A$、$R_4=R_6=R_B$，则：
			\begin{gather}
				R_B=(\sqrt{2}-1)R_A
			\end{gather}\par
			设置电阻阻值为$R_A=120$k$\Omega$(Standeard Value,$1\%$)，可计算得到$R_B=49.70562748\Omega$，与标准电阻值表比对后，实际选用电阻阻值为$R_B=49.9$k$\Omega$(Standeard Value,$1\%$)的电阻。\par
			综上所述，最终电阻阻值选用为$R_1=R_2=3.16$k$\Omega$、$R_3=R_5=120$k$\Omega$、$R_4=R_6=49.9$k$\Omega$
			将要求的信号接入该电路，得到仿真电路图，如图\ref{fig:二阶有源带通滤波器仿真原理图}所示。
			\begin{figure}[H]
				\centering
				\includegraphics[width=0.70\textwidth]{./12.第十二题示波器现象/12仿真原理图}
				\caption{二阶有源带通滤波器仿真原理图}
				\label{fig:二阶有源带通滤波器仿真原理图}
			\end{figure}\par
			对输入和输出信号做傅里叶变换，得到对应频谱曲线，如图\ref{fig:二阶有源带通滤波器频谱曲线}所示。
			\begin{figure}[H]
				\centering
				\includegraphics[width=0.45\textwidth]{./12.第十二题示波器现象/12仿真频率曲线}
				\caption{二阶有源带通滤波器频谱曲线}
				\label{fig:二阶有源带通滤波器频谱曲线}
			\end{figure}\par
			对于低频信号（200Hz正弦信号），幅值从0.399986降到了0.275028，可求得增益为$A=20\lg(\frac{0.275028}{0.399986})=-3.253357602$dB；对于中频信号（500Hz正弦信号），幅值从0.399954降到了0.400859，可求得增益为$A=20\lg(\frac{0.400859}{0.399954})=0.01963188277$dB；对于高频信号（1kHz正弦信号），幅值从0.0999493降到了0.0804759，可求得增益为$A=20\lg(\frac{0.0804759}{0.0999493})=-1.882278291$dB。\par
			这说明图\ref{fig:二阶有源带通滤波器电路原理图}的带通滤波器可以对高频信号和低频信号进行滤波操作。

	\subsection{结论与收获}
		\subsubsection{结论}
		12.1经过验证，所设计二阶有源带通滤波器可正常工作，仿真结果与预期结果误差再允许范围之内，由此判断所设计电路正确。\par
		12.2性能分析：在中频信号对应频率处，输出信号幅值有明显峰值且相对较高，而在其他频率处幅值显著降低，则表明该二阶有源带通滤波器能够提取题 6 [1] e) 中的中频信号，完成了设计功能。从性能方面来看通带宽度适中，过渡带较陡峭，对中频信号外的频率抑制效果良好。
